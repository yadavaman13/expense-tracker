# Expense Tracker Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an offline-first expense tracker mobile app with dashboard, transaction management, filtering, and analytics using Expo + React Native Paper.

**Architecture:** Feature-based folder structure with Expo Router file-based routing, Zustand for state management with AsyncStorage persistence, and React Native Paper with a minimalist B/W/grey theme.

**Tech Stack:** React Native, Expo (SDK 52+, managed), JavaScript, Expo Router, Zustand, React Native Paper, AsyncStorage, react-native-chart-kit

**Spec:** `docs/specs/2026-11-02-expense-tracker-design.md`

## Global Constraints

- Expo SDK 52+ managed workflow
- JavaScript (not TypeScript)
- No internet required — fully offline
- All colors from the B/W/grey palette defined in spec (green/red only for income/expense amounts)
- No gradients, no heavy shadows — elevation 1 max
- System default fonts only (no custom fonts)
- Touch targets ≥ 48dp for accessibility
- Transaction amounts must be > 0
- Dates must not be in the future

```js
// src/utils/calculations.js

/**
 * Calculate balance, total income, and total expenses from transactions.
 * @param {Array} transactions - Array of transaction objects
 * @returns {{ balance: number, totalIncome: number, totalExpense: number }}
 */
export function calculateBalance(transactions) {
  const totalIncome = transactions
    .filter((t) => t.type === "income")
    .reduce((sum, t) => sum + t.amount, 0);

  const totalExpense = transactions
    .filter((t) => t.type === "expense")
    .reduce((sum, t) => sum + t.amount, 0);

  return {
    balance: totalIncome - totalExpense,
    totalIncome,
    totalExpense,
  };
}

/**
 * Calculate spending totals per category.
 * @param {Array} transactions - Array of transaction objects
 * @param {"income"|"expense"} type - Transaction type to aggregate
 * @returns {Array<{ category: string, total: number, percentage: number }>}
 */
export function calculateCategoryTotals(transactions, type) {
  const filtered = transactions.filter((t) => t.type === type);
  const grandTotal = filtered.reduce((sum, t) => sum + t.amount, 0);

  if (grandTotal === 0) return [];

  const categoryMap = {};
  filtered.forEach((t) => {
    categoryMap[t.category] = (categoryMap[t.category] || 0) + t.amount;
  });

  return Object.entries(categoryMap)
    .map(([category, total]) => ({
      category,
      total,
      percentage: Math.round((total / grandTotal) * 100),
    }))
    .sort((a, b) => b.total - a.total);
}
```

```js
// src/utils/formatters.js

const MONTHS = [
  "Jan", "Feb", "Mar", "Apr", "May", "Jun",
  "Jul", "Aug", "Sep", "Oct", "Nov", "Dec",
];

const MONTHS_FULL = [
  "January", "February", "March", "April", "May", "June",
  "July", "August", "September", "October", "November", "December",
];

/**
 * Format a number as currency string.
 * @param {number} amount
 * @returns {string} e.g., "₹1,250.00"
 */
export function formatCurrency(amount) {
  return "₹" + amount.toLocaleString("en-IN", {
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  });
}

/**
 * Format an ISO date string to readable format.
 * @param {string} dateString - "2026-11-02"
 * @returns {string} e.g., "02 Nov 2026"
 */
export function formatDate(dateString) {
  const date = new Date(dateString + "T00:00:00");
  const day = String(date.getDate()).padStart(2, "0");
  const month = MONTHS[date.getMonth()];
  const year = date.getFullYear();
  return `${day} ${month} ${year}`;
}

/**
 * Format an ISO date string to short format.
 * @param {string} dateString - "2026-11-02"
 * @returns {string} e.g., "02 Nov"
 */
export function formatDateShort(dateString) {
  const date = new Date(dateString + "T00:00:00");
  const day = String(date.getDate()).padStart(2, "0");
  const month = MONTHS[date.getMonth()];
  return `${day} ${month}`;
}

/**
 * Get full month name from month index.
 * @param {number} monthIndex - 0-11
 * @returns {string} e.g., "November"
 */
export function getMonthName(monthIndex) {
  return MONTHS_FULL[monthIndex];
}
```

```js
// src/utils/filters.js

/**
 * Filter transactions by type, category, month/year, and search query.
 * @param {Array} transactions
 * @param {{ type?: string, category?: string, month?: number, year?: number, searchQuery?: string }} filters
 * @returns {Array} Filtered transactions
 */
export function filterTransactions(transactions, filters = {}) {
  let result = [...transactions];

  if (filters.type && filters.type !== "all") {
    result = result.filter((t) => t.type === filters.type);
  }

  if (filters.category && filters.category !== "all") {
    result = result.filter((t) => t.category === filters.category);
  }

  if (filters.month !== undefined && filters.year !== undefined) {
    result = result.filter((t) => {
      const date = new Date(t.date + "T00:00:00");
      return date.getMonth() === filters.month && date.getFullYear() === filters.year;
    });
  }

  if (filters.searchQuery && filters.searchQuery.trim()) {
    const query = filters.searchQuery.toLowerCase().trim();
    result = result.filter(
      (t) =>
        t.description.toLowerCase().includes(query) ||
        t.category.toLowerCase().includes(query)
    );
  }

  return result;
}

/**
 * Sort transactions by date.
 * @param {Array} transactions
 * @param {"desc"|"asc"} order - Sort order, defaults to "desc" (newest first)
 * @returns {Array} Sorted transactions (new array)
 */
export function sortTransactionsByDate(transactions, order = "desc") {
  return [...transactions].sort((a, b) => {
    const dateA = new Date(a.date);
    const dateB = new Date(b.date);
    return order === "desc" ? dateB - dateA : dateA - dateB;
  });
}
```

- [ ] **Step 4: Create the Zustand store**

```js
// src/store/useTransactionStore.js
import { create } from "zustand";
import { persist, createJSONStorage } from "zustand/middleware";
import AsyncStorage from "@react-native-async-storage/async-storage";
import { v4 as uuidv4 } from "uuid";

const useTransactionStore = create(
  persist(
    (set) => ({
      transactions: [],

      addTransaction: (data) => {
        const transaction = {
          id: uuidv4(),
          type: data.type,
          amount: parseFloat(data.amount),
          category: data.category,
          description: data.description || "",
          date: data.date,
          createdAt: new Date().toISOString(),
        };

        set((state) => ({
          transactions: [transaction, ...state.transactions],
        }));
      },

      updateTransaction: (id, data) => {
        set((state) => ({
          transactions: state.transactions.map((t) =>
            t.id === id
              ? {
                  ...t,
                  type: data.type !== undefined ? data.type : t.type,
                  amount: data.amount !== undefined ? parseFloat(data.amount) : t.amount,
                  category: data.category !== undefined ? data.category : t.category,
                  description: data.description !== undefined ? data.description : t.description,
                  date: data.date !== undefined ? data.date : t.date,
                }
              : t
          ),
        }));
      },

      deleteTransaction: (id) => {
        set((state) => ({
          transactions: state.transactions.filter((t) => t.id !== id),
        }));
      },
    }),
    {
      name: "expense-tracker-storage",
      storage: createJSONStorage(() => AsyncStorage),
    }
  )
);

export default useTransactionStore;
```

```js
          />
```
```js
        </View>
```

```js
              if (errors.category) setErrors({ ...errors, category: undefined });
            }}
```

```js
import { useState, useMemo } from "react";
import { filterTransactions, sortTransactionsByDate } from "../utils/filters";

const DEFAULT_FILTERS = {
  type: "all",
  category: "all",
  searchQuery: "",
};

export default function useFilteredTransactions(transactions) {
  const [filters, setFilters] = useState(DEFAULT_FILTERS);

  return { filteredTransactions, filters, setFilters, resetFilters };
}
```

```js
import { useMemo } from "react";
import { filterTransactions } from "../utils/filters";
import { calculateBalance, calculateCategoryTotals } from "../utils/calculations";

export default function useMonthlyStats(transactions, month, year) {
  return useMemo(() => {
    const monthlyTransactions = filterTransactions(transactions, { month, year });
    const { balance, totalIncome, totalExpense } = calculateBalance(monthlyTransactions);

      totalIncome,

      categoryTotals,
      topCategory,
      transactionCount: monthlyTransactions.length,
```
